# Berkas ini berisi pesan kesalahan yang ditampilkan kepada pengguna saat
# proses masuk melalui HTTP/HTTPS.
# Pesan dapat diubah agar lebih mudah dipahami oleh pengguna.
#
# Variabel yang paling sering digunakan:
#	$(error-orig)	- pesan kesalahan asli dari hotspot
#	$(ip)		- alamat IP pengguna
#	$(username)	- nama pengguna yang mencoba masuk

# internal-error
# Kesalahan internal yang seharusnya tidak terjadi. Variabel error-orig
# menjelaskan penyebab kesalahan dari hotspot.

internal-error = Terjadi kesalahan internal ($(error-orig)).

# config-error
# Kesalahan ini tidak akan muncul jika hotspot dikonfigurasi dengan benar.

config-error = Terjadi kesalahan konfigurasi ($(error-orig)).

# not-logged-in
# Muncul saat pengguna yang belum masuk membuka halaman status atau keluar.

not-logged-in = Anda belum masuk ke hotspot (IP $(ip)).

# ippool-empty
# Muncul saat tidak ada alamat IP yang tersedia di dalam IP pool.

ippool-empty = Alamat IP tidak dapat diberikan karena tidak ada alamat yang tersedia.

# shutting-down
# Muncul saat layanan hotspot sedang dimatikan dan tidak menerima pengguna baru.

shutting-down = Layanan hotspot sedang dimatikan. Silakan coba lagi nanti.

# user-session-limit
# Muncul saat jumlah perangkat aktif telah mencapai batas shared-users.

user-session-limit = Batas perangkat untuk pengguna $(username) telah tercapai.

# license-session-limit
# Muncul saat jumlah pengguna aktif telah mencapai batas lisensi MikroTik.

license-session-limit = Batas sesi lisensi telah tercapai ($(error-orig)).

# wrong-mac-username
# Muncul saat nama pengguna berbentuk alamat MAC, tetapi tidak cocok dengan
# alamat MAC perangkat yang sedang digunakan.

wrong-mac-username = Nama pengguna $(username) tidak valid karena alamat MAC perangkat tidak sesuai.

# chap-missing
# Muncul saat metode HTTP-CHAP digunakan, tetapi hotspot tidak menerima
# kata sandi terenkripsi. Pastikan JavaScript aktif, lalu coba masuk kembali.

chap-missing = Browser tidak mengirim respons autentikasi. Aktifkan JavaScript, lalu coba lagi.

# invalid-username
# Muncul saat nama pengguna atau kata sandi tidak valid. Pesan Access-Reject
# dari server RADIUS dapat menggantikan pesan ini.

invalid-username = Nama pengguna atau kata sandi salah. Silakan periksa kembali.

# invalid-mac
# Muncul saat akun lokal terikat ke alamat MAC lain.

invalid-mac = Pengguna $(username) hanya dapat digunakan pada perangkat yang terdaftar.

# uptime-limit, traffic-limit
# Muncul saat pengguna lokal mencapai batas waktu atau kuota data.

uptime-limit = Batas waktu penggunaan untuk pengguna $(username) telah habis.
traffic-limit = Batas kuota data untuk pengguna $(username) telah habis.

# radius-timeout
# Muncul saat server RADIUS tidak memberikan respons.

radius-timeout = Server RADIUS tidak merespons. Silakan coba lagi.

# auth-in-progress
# Muncul saat permintaan autentikasi sebelumnya masih diproses.

auth-in-progress = Proses masuk sedang berlangsung. Silakan coba lagi sebentar.

# radius-reply
# Pesan khusus yang dikirim oleh server RADIUS.

radius-reply = Pesan dari server autentikasi: $(error-orig)
